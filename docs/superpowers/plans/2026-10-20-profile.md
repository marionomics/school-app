> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Every username and avatar opens `/u/:username`, a profile with photo, bio, aura received, two tabs of posts, and — for the owner and the teacher only — the grade breakdown.

**Architecture:** The feed's class-scoping rule moves into `app/services/visibility.py` so the feed and profiles share it. A new `app/routers/profiles.py` serves the header and paginated posts. The frontend gets a `/u/:username` page built from extracted components (`GradeBreakdown`, `ProfileEditor`, `RemovePhotoButton`) and a `UserLink` used everywhere a username or avatar is printed.

**Tech Stack:** FastAPI + SQLAlchemy + pytest; React + Vite + TanStack Query; shadcn `base-sera` on `@base-ui/react`; `@remixicon/react`.

**Spec:** `docs/superpowers/specs/2026-10-20-profile-design.md` — read it before starting.

## Global Constraints

- A grade is visible only to its owner and to a teacher who teaches that student. Everyone else receives **no `grades` key** (not an empty list).
- Grade summaries come from `app.routers.grades._summary(db, user_id, klass)` — never a second grade computation.
- `aura_received` = `SUM(posts.like_count)` over the person's posts with `status = 'active'` (deleted and vetoed excluded).
- Entregas (`is_entrega = true`) never appear on a profile.
- Profile lists: `status IN ('active','vetoed')`, newest first by `(created_at, id)`, keyset cursor via `app.services.cursor.encode_cursor/decode_cursor`, page size 20 (max 50).
- A tarea/examen of a class the viewer neither attends (any enrollment status) nor teaches never appears — nor, in Respuestas, a reply whose **root** is one.
- `get_feed` behaviour must not change; `backend/tests/test_feed_scoping.py` and the other feed tests are the guard.
- All copy in `frontend/src/strings/es.ts`, Spanish placeholders, **tú** form (never usted). Every points value through `formatPoints` (`frontend/src/lib/points.ts`). Icons only from `@remixicon/react`. Preset components unmodified. One control per line in any form. Mobile first.
- Every username or avatar printed in the UI goes through `UserLink` (Task 4). No link nested inside another link.
- `src/lib` modules with a `*.test.ts` use relative imports only.
- Backend tests: `cd backend && .venv/bin/python -m pytest -q`. Frontend: `cd frontend && npx tsc -b --noEmit && npx eslint src && npm test && npm run build` (one pre-existing eslint warning in `src/components/ui/button.tsx` — leave it).

```python
from typing import List

from sqlalchemy import or_
from sqlalchemy.orm import Session

from app.models import Class, Enrollment, Post, User

ASSIGNMENT_TYPES = ("tarea", "examen")


def viewer_class_ids(db: Session, viewer: User) -> List[int]:
    """Classes whose assignments this viewer may see: every enrollment —
    no status filter on purpose, ghost and polizón are in the class — plus
    every class they teach."""
    ids = [cid for (cid,) in db.query(Enrollment.class_id)
           .filter(Enrollment.user_id == viewer.id).all()]
    ids += [cid for (cid,) in db.query(Class.id)
            .filter(Class.teacher_id == viewer.id).all()]
    return ids


def assignment_visible(class_ids: List[int]):
    """Social content is global — all classes, all semesters. Assignments are
    not: a student cannot tell another class's tarea from work they owe."""
    return or_(Post.type.notin_(ASSIGNMENT_TYPES), Post.class_id.in_(class_ids))


def root_visible(root: Post, class_ids: List[int]) -> bool:
    """The same rule, for a thread's root already in memory."""
    return root.type not in ASSIGNMENT_TYPES or root.class_id in class_ids
```

```python
from typing import List, Optional

from fastapi import APIRouter, Depends, HTTPException, Query
from sqlalchemy import and_, func, or_
from sqlalchemy.orm import Session, selectinload

from app.auth.deps import get_current_user
from app.database import get_db
from app.models import Class, Enrollment, Post, User
from app.routers.grades import _summary
from app.routers.posts import _liked_ids, _teacher_aura_ids, get_root, serialize_post
from app.services.cursor import decode_cursor, encode_cursor
from app.services.media import avatar_for
from app.services.visibility import assignment_visible, root_visible, viewer_class_ids

router = APIRouter(prefix="/api/users", tags=["profiles"])

EXCERPT = 120


def _user_or_404(db: Session, username: str) -> User:
    u = db.query(User).filter(User.username == username).first()
    if u is None:
        raise HTTPException(status_code=404, detail="Ese usuario no existe")
    return u


def _visible_grades(db: Session, viewer: User, u: User) -> Optional[List[dict]]:
    """A grade is seen by its owner and by a teacher who teaches that student
    — nobody else. None means the key is left out entirely."""
    base = db.query(Class).join(Enrollment, Enrollment.class_id == Class.id) \
        .filter(Enrollment.user_id == u.id)
    if viewer.id == u.id:
        classes = base.filter(Enrollment.status == "active").order_by(Class.id).all()
    elif viewer.role == "teacher":
        classes = base.filter(Class.teacher_id == viewer.id).order_by(Class.id).all()
        if not classes:
            return None
    else:
        return None
    return [{"class_id": k.id, "class_name": k.name, "summary": _summary(db, u.id, k)}
            for k in classes]


@router.get("/{username}")
def profile(username: str, viewer: User = Depends(get_current_user),
            db: Session = Depends(get_db)):
    u = _user_or_404(db, username)
    aura = (
        db.query(func.coalesce(func.sum(Post.like_count), 0))
        .filter(Post.author_id == u.id, Post.status == "active")
        .scalar()
    )
    out = {
        "id": u.id, "username": u.username, "name": u.name, "bio": u.bio,
        "avatar_url": avatar_for(u), "role": u.role, "is_me": viewer.id == u.id,
        "aura_received": int(aura), "has_own_avatar": u.has_own_avatar,
    }
    grades = _visible_grades(db, viewer, u)
    if grades is not None:
        out["grades"] = grades
    return out


@router.get("/{username}/posts")
def profile_posts(username: str,
                  tab: str = Query("posts", pattern="^(posts|replies)$"),
                  cursor: Optional[str] = Query(None),
                  limit: int = Query(20, ge=1, le=50),
                  viewer: User = Depends(get_current_user),
                  db: Session = Depends(get_db)):
    u = _user_or_404(db, username)
    class_ids = viewer_class_ids(db, viewer)
    q = (
        db.query(Post)
        .options(selectinload(Post.author), selectinload(Post.klass),
                 selectinload(Post.attachments))
        .filter(Post.author_id == u.id,
                Post.status.in_(("active", "vetoed")),
                assignment_visible(class_ids))
    )
    if tab == "posts":
        q = q.filter(Post.parent_id.is_(None))
    else:
        # Entregas live in their tarea's thread; listing them per student
        # would make homework easy to browse and copy.
        q = q.filter(Post.parent_id.isnot(None), Post.is_entrega.is_(False))
    if cursor:
        try:
            c_dt, c_id = decode_cursor(cursor)
        except Exception:
            raise HTTPException(status_code=422, detail="Cursor inválido")
        q = q.filter(or_(Post.created_at < c_dt,
                         and_(Post.created_at == c_dt, Post.id < c_id)))
    rows = q.order_by(Post.created_at.desc(), Post.id.desc()).limit(limit + 1).all()
    has_more = len(rows) > limit
    rows = rows[:limit]
    # The cursor comes from the last FETCHED row, so a page whose rows were all
    # filtered out below still lets the client reach older ones.
    next_cursor = encode_cursor(rows[-1].created_at, rows[-1].id) if (has_more and rows) else None

    kept = []
    for p in rows:
        if tab == "replies" and not root_visible(get_root(db, p), class_ids):
            continue
        kept.append(p)

    ids = [p.id for p in kept]
    liked = _liked_ids(db, viewer, ids)
    aura_ids = _teacher_aura_ids(db, ids)
    items = []
    for p in kept:
        d = serialize_post(p, liked, viewer=viewer, teacher_aura_ids=aura_ids).model_dump()
        if tab == "replies":
            parent = db.get(Post, p.parent_id)
            d["parent"] = {
                "id": parent.id,
                "author_username": parent.author.username,
                "excerpt": "" if parent.status == "deleted" else parent.content[:EXCERPT],
            }
        items.append(d)
    return {"items": items, "next_cursor": next_cursor}
```

```tsx
function PostList({ username, tab }: { username: string; tab: Tab }) {
  const { user } = useAuth();
  const like = useLikeMutation();
  const veto = useVetoFlow();

  const q = useInfiniteQuery({
    queryKey: ["profile-posts", username, tab],

      if (entries[0].isIntersecting && q.hasNextPage && !q.isFetchingNextPage) {
        void q.fetchNextPage();
      }
    });
    obs.observe(el);
    return () => obs.disconnect();
  }, [q]);

  if (q.isPending) return <FeedSkeleton />;
```

```tsx
import { Link } from "react-router-dom";
import Avatar from "@/components/Avatar";
import { cn } from "@/lib/utils";
import type { Author } from "@/lib/types";

type Who = Pick<Author, "username" | "name"> & Partial<Author>;

/** Every username or avatar in the app goes through here (CLAUDE.md): if a
 *  name appears, you can tap it and reach the profile. A user who has not
 *  picked a username yet renders as plain text, never a broken link. */
export default function UserLink({
  user,
  avatar = false,
  size = 40,
  className,
}: {
  user: Who;
  avatar?: boolean;
  size?: number;
  className?: string;
}) {
  const body = avatar ? (
    <Avatar author={user as Author} size={size} />
  ) : (
    <span>@{user.username ?? user.name}</span>
  );
  if (!user.username) return <span className={className}>{body}</span>;
  return (
    <Link
      to={`/u/${user.username}`}
      className={cn("hover:underline", className)}
      // Inside clickable rows (/revisar) the tap opens the profile only.
      onClick={(e) => e.stopPropagation()}
    >
      {body}
    </Link>
  );
}
```
